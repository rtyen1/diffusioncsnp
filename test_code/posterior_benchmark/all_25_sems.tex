\noindent All models share mean $\mu=(0,0,0)$; the noises $\varepsilon_A,\varepsilon_B,\varepsilon_C$
are mutually independent Gaussians. DAGs in the same MEC share the same fixed covariance $\Sigma_c$
and hence the same joint Gaussian $N(0,\Sigma_c)$.
\paragraph{MEC 1} Shared covariance \(\Sigma_1=\begin{pmatrix}1 & 0 & 0 \\0 & 1 & 0 \\0 & 0 & 1 \\\end{pmatrix}\); DAGs: \emph{empty}.

\noindent\textbf{DAG \#1: empty \ (MEC 1)}
\begin{align*}
A &= \varepsilon_A, & \varepsilon_A \sim \mathcal{N}(0, 1),\\B &= \varepsilon_B, & \varepsilon_B \sim \mathcal{N}(0, 1),\\C &= \varepsilon_C, & \varepsilon_C \sim \mathcal{N}(0, 1).
\end{align*}
